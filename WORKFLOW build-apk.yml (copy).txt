name: Build Inkwell for Android

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Check out the project
        uses: actions/checkout@v4

      - name: Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build the APK
        run: gradle --no-daemon assembleRelease

      - name: Name the APK
        run: cp app/build/outputs/apk/release/app-release.apk "Inkwell-7.29-build${{ github.run_number }}.apk"

      - name: Keep the APK with this run
        uses: actions/upload-artifact@v4
        with:
          name: Inkwell-APK
          path: Inkwell-*.apk

      - name: Publish a release you can open on the tablet
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh release create "build-${{ github.run_number }}" Inkwell-*.apk \
            --title "Inkwell 7.29 (build ${{ github.run_number }})" \
            --notes "On the Fire tablet: open this page in Silk, tap the .apk file under Assets, then Install. Each build installs over the previous one and keeps your documents."
